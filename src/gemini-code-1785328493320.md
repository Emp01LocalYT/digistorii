const formatDate = (dateStr?: string) => {
  if (!dateStr) return "N/A";
  const d = new Date(dateStr);
  return isNaN(d.getTime()) ? "N/A" : d.toLocaleDateString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric"
  });
};

useEffect(() => {

    const expired = diffDays <= 0 || subscription.status?.toLowerCase() === "expired";

  cs.plan_id,
  cs.subscription_end,
  cs.status,
  cs.billing_interval,
  p.plan_name

LEFT JOIN public.payments p ON p.subscription_id = cs.id

ORDER BY cs.updated_at DESC LIMIT 1;
